function addToCart(id){

  const p=PRODUTOS.find(x=>x.id===id); if(!p)return;

  const item=carrinho.find(x=>x.id===id);

  if(item)item.qtd++; else carrinho.push({...p,qtd:1});

  renderCart();openCart();

}

function changeQty(id,delta){

  const item=carrinho.find(x=>x.id===id);if(!item)return;

  item.qtd+=delta;

  if(item.qtd<=0)carrinho=carrinho.filter(x=>x.id!==id);

  renderCart();

}

function renderCart(){

  const list=document.getElementById("cartList");

  const count=carrinho.reduce((a,x)=>a+x.qtd,0);

  const total=carrinho.reduce((a,x)=>a+x.preco*x.qtd,0);

  document.getElementById("cartCount").textContent=count;

  document.getElementById("cartTotal").textContent=money(total);

  if(!carrinho.length){list.innerHTML='<div class="empty">Seu pedido está vazio.</div>';return}

  list.innerHTML=carrinho.map(i=>`

    <div class="cart-row">

      <div><strong>${escapeHtml(i.nome)}</strong><br><span style="color:var(--suave);font-size:12px">${money(i.preco)} cada</span></div>

      <div class="qty"><button onclick="changeQty(${i.id},-1)">−</button><strong>${i.qtd}</strong><button onclick="changeQty(${i.id},1)">+</button></div>

    </div>`).join("");

}

function checkout(){

  if(!carrinho.length){alert("Seu pedido está vazio.");return}

  let msg=`NOVO PEDIDO — ${CONFIG.nomeLoja}\n\n`;

  carrinho.forEach(i=>{msg+=`• ${i.nome} — ${i.qtd}x — ${money(i.preco*i.qtd)}\n`});

  const total=carrinho.reduce((a,x)=>a+x.preco*x.qtd,0);

  msg+=`\nTotal dos produtos: ${money(total)}\n\n`;

  msg+="Olá! Fiz este pedido pelo site da Mater Dei. Gostaria de confirmar disponibilidade, pagamento e envio.";

  window.open(wa(msg),"_blank");

}
